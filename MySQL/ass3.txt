CREATE TABLE studmarks (
  Name VARCHAR(20) NOT NULL,
  TotalMarks INT NOT NULL
);

CREATE TABLE result (
  Roll_No TINYINT NOT NULL,
  Name VARCHAR(20) NOT NULL,
  Class VARCHAR(25) NOT NULL
);

-- Step 4: Create procedure to categorize students
DELIMITER $$

CREATE PROCEDURE proc_grade(
  IN rollno TINYINT,
  IN name VARCHAR(20),
  IN marks INT
)
BEGIN
  DECLARE class VARCHAR(25);

  -- Grading logic
  IF marks >= 990 AND marks <= 1500 THEN
    SET class = 'Distinction';
  ELSEIF marks <= 989 AND marks >= 900 THEN
    SET class = 'First Class';
  ELSEIF marks <= 899 AND marks >= 825 THEN
    SET class = 'Second Class';
  ELSEIF marks <= 824 AND marks >= 700 THEN
    SET class = 'Pass';
  ELSE
    SET class = 'Fail';
  END IF;

  -- Insert records into both tables
  INSERT INTO studmarks VALUES (name, marks);
  INSERT INTO result VALUES (rollno, name, class);
END $$

DELIMITER ;

CALL proc_grade(3, 'Smith', 834);
CALL proc_grade(4, 'Carol', 750);
CALL proc_grade(5, 'Bob', 950);
CALL proc_grade(6, 'Sam', 650);

-- Step 7: Create function to return total students in a given class
DELIMITER $$

CREATE FUNCTION tot_stud(classname VARCHAR(25))
RETURNS INT
DETERMINISTIC
BEGIN
  DECLARE total INT;
  SELECT COUNT(*) INTO total
  FROM result
  WHERE Class = classname;
  RETURN total;
END $$

DELIMITER ;
