PL/SQL Stored Procedure and Stored Function. Create the following tables. • Stud_Marks(name, total_marks) • Result(Roll,Name, Class) Write a Stored Procedure namely proc_Grade for the categorization of students. If marks scored by a student in examination is <=1500 and marks>=990 then a student will be placed in distinction category if marks scored are between 989 and 900 category is first class, if marks 899 and 825 category is Higher Second Class otherwise fail. 1. Write a PL/SQL block to use the procedure created with the above requirement. Insert the data in both the tables by calling the above procedure. 2. Write a function which will return the total students in a given class.

CREATE DATABASE student_grades_db;
USE student_grades_db;
CREATE TABLE studmarks (
  Name VARCHAR(20) NOT NULL,
  TotalMarks INT NOT NULL
);
CREATE TABLE result (
  Roll_No TINYINT NOT NULL,
  Name VARCHAR(20) NOT NULL,
  Class VARCHAR(25) NOT NULL
);
DELIMITER $$

CREATE PROCEDURE proc_grade(
  IN rollno TINYINT,
  IN name VARCHAR(20),
  IN marks INT
)
BEGIN
  DECLARE class VARCHAR(25);
IF marks >= 990 AND marks <= 1500 THEN
    SET class = 'Distinction';
  ELSEIF marks <= 989 AND marks >= 900 THEN
    SET class = 'First Class';
  ELSEIF marks <= 899 AND marks >= 825 THEN
    SET class = 'Second Class';
  ELSEIF marks <= 824 AND marks >= 700 THEN
    SET class = 'Pass';
  ELSE
    SET class = 'Fail';
  END IF;
INSERT INTO studmarks VALUES (name, marks);
  INSERT INTO result VALUES (rollno, name, class);
END $$

DELIMITER ;
CALL proc_grade(1, 'Aryan', 850);
CALL proc_grade(2, 'Peter', 1000);
CALL proc_grade(3, 'Smith', 834);
CALL proc_grade(4, 'Carol', 750);
CALL proc_grade(5, 'Bob', 950);
CALL proc_grade(6, 'Sam', 650);
SELECT * FROM result;
SELECT * FROM studmarks;
DELIMITER $$

CREATE FUNCTION tot_stud(classname VARCHAR(25))
RETURNS INT
DETERMINISTIC
BEGIN
  DECLARE total INT;
  SELECT COUNT(*) INTO total
  FROM result
  WHERE Class = classname;
  RETURN total;
END $$

DELIMITER ;
SELECT tot_stud('Second Class') AS 'Second Class Students';
SELECT tot_stud('Pass') AS 'Pass Students';
SELECT tot_stud('Distinction') AS 'Distinction Students';
SELECT tot_stud('Fail') AS 'Fail Students';
